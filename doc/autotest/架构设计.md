自动化测试平台 - 完整实现
基于自研框架，提供可运行的自动化测试平台，包含完整的数据库、后端Java服务、前端HTML/JS页面。

1. 完整 SQL 文件 (autotest.sql)
   所有表使用 CHAR(36) 作为主键，由应用层生成 UUID，不在数据库层自动生成。

StepExecutor.java（模板类）
java
package com.omni.autotest.core.engine;

import com.omni.autotest.core.model.TestStep;
import com.omni.autotest.core.model.TestAssertion;
import com.omni.framework.log.Logger;
import com.omni.framework.log.LoggerFactory;

public abstract class StepExecutor {
protected static final Logger log = LoggerFactory.getLogger(StepExecutor.class);

    public final StepResult execute(TestStep step, Context context, List<TestAssertion> assertions) {
        StepResult result = new StepResult(step.getId(), step.getStepOrder());
        long start = System.currentTimeMillis();
        try {
            beforeExecute(step, context);
            Object actionResult = doExecute(step, context);
            result.setActionResult(actionResult);
            afterExecute(step, context, actionResult);
            // execute assertions
            for (TestAssertion assertion : assertions) {
                if (assertion.getStepId() == null || assertion.getStepId().equals(step.getId())) {
                    AssertionResult ar = executeAssertion(assertion, context);
                    result.getAssertionResults().add(ar);
                    if (!ar.isPassed() && assertion.getFailHandle() == 1) {
                        result.setPassed(false);
                        break;
                    }
                }
            }
            result.setPassed(true);
        } catch (Exception e) {
            log.error("Step execution failed", e);
            result.setPassed(false);
            result.setError(e);
            handleFailure(step, context, e);
        }
        result.setDurationMs(System.currentTimeMillis() - start);
        return result;
    }

    protected void beforeExecute(TestStep step, Context context) throws Exception {
        // 处理等待
        if ("sleep".equals(step.getWaitType()) && step.getWaitSeconds() > 0) {
            Thread.sleep(step.getWaitSeconds() * 1000L);
        }
        // 变量替换
        if (step.getActionValue() != null) {
            step.setActionValue(VariableResolver.resolve(step.getActionValue(), context.getVariables()));
        }
    }

    protected abstract Object doExecute(TestStep step, Context context) throws Exception;

    protected void afterExecute(TestStep step, Context context, Object result) throws Exception {
        // 可提取变量
    }

    protected AssertionResult executeAssertion(TestAssertion assertion, Context context) throws Exception {
        return AssertionHandlerRegistry.getHandler(assertion.getAssertType()).execute(assertion, context);
    }

    protected void handleFailure(TestStep step, Context context, Exception e) {
        if (step.getFailHandle() == 1) {
            context.stopped = true;
        }
    }
}

Pipeline.java
java
package com.omni.autotest.core.engine;

import com.omni.autotest.core.model.TestCase;
import com.omni.autotest.core.model.TestStep;
import com.omni.autotest.mapper.TestStepMapper;
import com.omni.autotest.mapper.TestAssertionMapper;
import com.omni.framework.spring.context.SpringContextHolder;

import java.util.List;

public class Pipeline {
public static ExecutionResult execute(TestCase testCase, Context context) {
TestStepMapper stepMapper = SpringContextHolder.getApplicationContext().getBeanFactory().getBean(TestStepMapper.class);
TestAssertionMapper assertionMapper = SpringContextHolder.getApplicationContext().getBeanFactory().getBean(TestAssertionMapper.class);

        List<TestStep> steps = stepMapper.selectByCaseId(testCase.getId());
        List<TestAssertion> assertions = assertionMapper.selectByCaseId(testCase.getId());

        ExecutionResult execResult = new ExecutionResult(testCase.getId());
        for (TestStep step : steps) {
            if (context.stopped) break;
            StepExecutor executor = StepExecutorRegistry.getExecutor(step.getActionType());
            if (executor == null) {
                throw new RuntimeException("No executor for action: " + step.getActionType());
            }
            StepResult stepResult = executor.execute(step, context, assertions);
            execResult.addStepResult(stepResult);
        }
        return execResult;
    }
}

package com.omni.autotest.service;

import com.omni.autotest.core.engine.Context;
import com.omni.autotest.core.engine.ExecutionResult;
import com.omni.autotest.core.engine.Pipeline;
import com.omni.autotest.core.model.TestCase;
import com.omni.autotest.core.model.TestExecution;
import com.omni.autotest.mapper.TestExecutionMapper;
import com.omni.framework.spring.annotation.Autowired;
import com.omni.framework.spring.annotation.Service;

import java.util.Date;
import java.util.UUID;

@Service
public class TestExecutionService {

    @Autowired
    private TestCaseService testCaseService;

    @Autowired
    private TestExecutionMapper executionMapper;

    public TestExecution execute(String caseId, String triggerUser) {
        TestCase testCase = testCaseService.getById(caseId);
        Context context = new Context();
        // 加载数据集等
        ExecutionResult result = Pipeline.execute(testCase, context);

        TestExecution execution = new TestExecution();
        execution.setId(UUID.randomUUID().toString());
        execution.setCaseId(caseId);
        execution.setExecutionCode("EXEC-" + System.currentTimeMillis());
        execution.setTriggerUser(triggerUser);
        execution.setStartTime(new Date(result.getStartTime()));
        execution.setEndTime(new Date(System.currentTimeMillis()));
        execution.setTotalSteps(result.getStepResults().size());
        long passCount = result.getStepResults().stream().filter(StepResult::isPassed).count();
        execution.setPassCount((int) passCount);
        execution.setFailCount(result.getStepResults().size() - (int) passCount);
        execution.setPassRate(execution.getTotalSteps() > 0 ? (passCount * 100.0 / execution.getTotalSteps()) : 0);
        execution.setStatus(execution.getFailCount() == 0 ? 1 : 2);
        execution.setTotalTimeMs((int)(System.currentTimeMillis() - result.getStartTime()));

        executionMapper.insert(execution);
        return execution;
    }

    public List<TestExecution> listByCase(String caseId) {
        return executionMapper.selectByCaseId(caseId);
    }
}
