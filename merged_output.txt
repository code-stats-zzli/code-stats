package com.codestats.business.assistance.table.mysql.controller;

import com.framework.jdbc.metedata.datamodel.DataModel;
import com.framework.jdbc.metedata.service.DatabaseMetadataService;

import com.framework.spring.annotation.Autowired;
import com.framework.spring.annotation.Controller;
import com.framework.spring.annotation.RequestMapping;
import com.framework.spring.annotation.ResponseBody;
import com.framework.spring.mvc.RequestContextHolder;
import com.framework.tomcat.connector.Request;
import com.framework.util.DataModelConverter;
import com.framework.util.JsonUtil;

import java.net.URLDecoder;
import java.util.*;

@Controller
@RequestMapping("/dbclient")
public class DatabaseClientController {

    @Autowired
    private DatabaseMetadataService metadataService;

    @RequestMapping("/api/databases")
    @ResponseBody
    public String apiDatabases() {
        try {
            DataModel dm = metadataService.getDatabasesAsDataModel();
            // 只返回数据库名称列表
            List<String> dbNames = new ArrayList<>();
            for (int i = 0; i < dm.getRowCount(); i++) {
                dbNames.add(dm.getRow(i).getValue(0).toString());
            }
            return JsonUtil.toJson(dbNames);
        } catch (Exception e) {
            return buildErrorJson(e.getMessage());
        }
    }

    @RequestMapping("/api/tables")
    @ResponseBody
    public String apiTables() {
        String dbName = getQueryParam("db");
        if (dbName == null || dbName.isEmpty()) {
            return buildErrorJson("Missing parameter: db");
        }
        try {
            DataModel dm = metadataService.getTablesAsDataModel(dbName, null, "%");
            // 只返回表名和类型两个字段
            Set<String> columns = new LinkedHashSet<>(Arrays.asList("TABLE_NAME", "TABLE_TYPE"));
            // 注意：DataModelConverter.rowsToJsonArray 返回的是 JSON 数组字符串，其中每行的键是原始列名
            // 前端期望 name/type，因此我们可以用别名映射，这里直接转换后通过 DataModelConverter 自定义列实现
            // 但 DataModelConverter 直接使用列名输出，为匹配前端，我们用 rowsToJsonArray 全字段再手动改？不，前端已适配。
            // 实际上前端 fetch 后直接访问 .name 和 .type，所以我们需要将列名映射一下。
            // 最简单的方式：直接构建 List<Map> 并序列化。
            List<Map<String, Object>> list = new ArrayList<>();
            for (int i = 0; i < dm.getRowCount(); i++) {
                Map<String, Object> item = new LinkedHashMap<>();
                item.put("name", dm.getRow(i).getValue("TABLE_NAME"));
                item.put("type", dm.getRow(i).getValue("TABLE_TYPE"));
                list.add(item);
            }
            return JsonUtil.toJson(list);
        } catch (Exception e) {
            return buildErrorJson(e.getMessage());
        }
    }

    @RequestMapping("/api/columns")
    @ResponseBody
    public String apiColumns() {
        String dbName = getQueryParam("db");
        String tableName = getQueryParam("table");
        if (dbName == null || tableName == null) {
            return buildErrorJson("Missing parameter: db or table");
        }
        try {
            DataModel dm = metadataService.getColumnsAsDataModel(dbName, null, tableName);
            // 使用 DataModelConverter 自定义需要返回的列
            Set<String> columns = new LinkedHashSet<>(Arrays.asList(
                    "COLUMN_NAME", "TYPE_NAME", "COLUMN_SIZE", "NULLABLE", "IS_AUTOINCREMENT", "REMARKS"
            ));
            return DataModelConverter.rowsToJsonArray(dm, columns);
        } catch (Exception e) {
            return buildErrorJson(e.getMessage());
        }
    }

    private String buildErrorJson(String message) {
        Map<String, Object> error = new LinkedHashMap<>();
        error.put("error", message);
        return JsonUtil.toJson(error);
    }

    private String getQueryParam(String name) {
        Request request = RequestContextHolder.getRequest();
        if (request == null) return null;
        String query = request.getQueryString();
        if (query == null || query.isEmpty()) return null;
        Map<String, String> params = parseQueryString(query);
        return params.get(name);
    }

    private Map<String, String> parseQueryString(String query) {
        Map<String, String> map = new HashMap<>();
        String[] pairs = query.split("&");
        for (String pair : pairs) {
            int idx = pair.indexOf("=");
            if (idx > 0) {
                try {
                    String key = URLDecoder.decode(pair.substring(0, idx), "UTF-8");
                    String value = URLDecoder.decode(pair.substring(idx + 1), "UTF-8");
                    map.put(key, value);
                } catch (Exception ignored) {}
            }
        }
        return map;
    }
}

public class CodestatsEngine {

    private static final String GUI_FLAG = "--gui";

    public static void run(Class<?> source, String[] args) {
        Config config = ConfigLoader.load(args);
        // 解析args，判断是否包含 --gui 标志（注意：ConfigLoader不处理该标志，需单独判断）

        for (String arg : args) {
            if (GUI_FLAG.equals(arg)) {
                guiMode = false;
                break;
            }
        }
        Launcher launcher = guiMode ? new GuiLauncher() : new ConsoleLauncher();
        try {
            launcher.launch(config);
        } catch (Exception e) {
            System.err.println("Launch failed: " + e.getMessage());
            e.printStackTrace();
            System.exit(1);
        }
    }

        // 无参数时默认使用控制台模式
        new ConsoleLauncher().launch(config);
    }

package com.codestats.lib.codestats.collector;

import com.codestats.lib.codestats.filter.CompositeFilter;
import java.io.IOException;
import java.nio.file.*;
import java.nio.file.attribute.BasicFileAttributes;
import java.util.concurrent.ConcurrentLinkedQueue;
import java.util.concurrent.atomic.AtomicInteger;

public class FileCollector {
    private final CompositeFilter filterChain;

    public FileCollector(CompositeFilter filterChain) {
        this.filterChain = filterChain;
    }

    public ConcurrentLinkedQueue<Path> collect(Path rootDir) throws IOException {
        ConcurrentLinkedQueue<Path> queue = new ConcurrentLinkedQueue<>();
        AtomicInteger fileCount = new AtomicInteger(0);
        Files.walkFileTree(rootDir, new SimpleFileVisitor<Path>() {
            @Override
            public FileVisitResult visitFile(Path file, BasicFileAttributes attrs) {
                if (filterChain.accept(file)) {
                    queue.add(file);
                    fileCount.incrementAndGet();
                }
                return FileVisitResult.CONTINUE;
            }
            @Override
            public FileVisitResult visitFileFailed(Path file, IOException exc) {
                System.err.println("Skip file: " + file + " - " + exc.getMessage());
                return FileVisitResult.CONTINUE;
            }
        });
        System.out.println("Collected " + fileCount.get() + " files.");
        return queue;
    }
}

package com.codestats.lib.codestats.config;

import org.yaml.snakeyaml.Yaml;
import java.io.FileInputStream;
import java.io.IOException;
import java.io.InputStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.Arrays;
import java.util.List;
import java.util.Map;

public class ConfigLoader {
    public static Config load(String[] args) {
        Config.Builder builder = new Config.Builder();
        // 1. 加载默认配置文件 config.yaml（优先类路径，其次文件系统）
        loadYamlFromClasspathOrFile("config.yaml", builder);
        // 2. 加载环境变量
        loadEnv(builder);
        // 3. 解析命令行参数（覆盖）
        parseArgs(args, builder);
        // 根目录回退逻辑
        if (builder.getRootDir() == null) {
            if (builder.getDefaultInputPath() != null) {
                builder.rootDir(builder.getDefaultInputPath());
            } else {
                builder.rootDir(Paths.get("."));
            }
        }
        return builder.build();
    }

    /**
     * 优先从类路径加载，若失败则从工作目录加载指定的 YAML 文件。
     */
    private static void loadYamlFromClasspathOrFile(String resourceName, Config.Builder builder) {
        // 1. 尝试从类路径读取（支持 resources 目录和 jar 内）
        try (InputStream is = ConfigLoader.class.getClassLoader().getResourceAsStream(resourceName)) {
            if (is != null) {
                parseYaml(is, builder);
                return;
            }
        } catch (IOException ignored) {}

        // 2. 类路径没找到，尝试文件系统
        Path filePath = Paths.get(resourceName);
        if (Files.exists(filePath)) {
            try (FileInputStream fis = new FileInputStream(filePath.toFile())) {
                parseYaml(fis, builder);
            } catch (IOException ignored) {}
        }
    }

    private static void parseYaml(InputStream inputStream, Config.Builder builder) {
        Yaml yaml = new Yaml();
        Map<String, Object> map = yaml.load(inputStream);
        // 路径字段统一处理：将反斜杠替换为正斜杠，避免 YAML 转义问题
        if (map.containsKey("rootDir")) {
            String dir = map.get("rootDir").toString().replace("\\", "/");
            builder.rootDir(Paths.get(dir));
        }
        if (map.containsKey("extensions")) builder.extensions((List<String>) map.get("extensions"));
        if (map.containsKey("excludePaths")) builder.excludePaths((List<String>) map.get("excludePaths"));
        if (map.containsKey("excludeComments")) builder.excludeComments((Boolean) map.get("excludeComments"));
        if (map.containsKey("threadCount")) builder.threadCount((Integer) map.get("threadCount"));
        if (map.containsKey("sortBy")) builder.sortBy((String) map.get("sortBy"));
        if (map.containsKey("outputFormat")) builder.outputFormat((String) map.get("outputFormat"));
        if (map.containsKey("outputFile")) {
            String f = map.get("outputFile").toString().replace("\\", "/");
            builder.outputFile(Paths.get(f));
        }
        if (map.containsKey("useGitIgnore")) builder.useGitIgnore((Boolean) map.get("useGitIgnore"));
        if (map.containsKey("defaultInputPath")) {
            String d = map.get("defaultInputPath").toString().replace("\\", "/");
            builder.defaultInputPath(Paths.get(d));
        }
        if (map.containsKey("defaultOutputPath")) {
            String d = map.get("defaultOutputPath").toString().replace("\\", "/");
            builder.defaultOutputPath(Paths.get(d));
        }
    }

    private static void loadEnv(Config.Builder builder) {
        String root = System.getenv("CODE_STATS_ROOT_DIR");
        if (root != null) builder.rootDir(Paths.get(root));
        String exts = System.getenv("CODE_STATS_EXTENSIONS");
        if (exts != null) builder.extensions(Arrays.asList(exts.split(",")));
        String threads = System.getenv("CODE_STATS_THREADS");
        if (threads != null) builder.threadCount(Integer.parseInt(threads));
        String sort = System.getenv("CODE_STATS_SORT_BY");
        if (sort != null) builder.sortBy(sort);
        String defaultInput = System.getenv("CODE_STATS_DEFAULT_INPUT");
        if (defaultInput != null) builder.defaultInputPath(Paths.get(defaultInput));
        String defaultOutput = System.getenv("CODE_STATS_DEFAULT_OUTPUT");
        if (defaultOutput != null) builder.defaultOutputPath(Paths.get(defaultOutput));
    }

    private static void parseArgs(String[] args, Config.Builder builder) {
        for (int i = 0; i < args.length; i++) {
            switch (args[i]) {
                case "-d": builder.rootDir(Paths.get(args[++i].replace("\\", "/"))); break;
                case "-e": builder.extensions(Arrays.asList(args[++i].split(","))); break;
                case "-x": builder.excludePaths(Arrays.asList(args[++i].split(","))); break;
                case "-c": builder.excludeComments(Boolean.parseBoolean(args[++i])); break;
                case "-t": builder.threadCount(Integer.parseInt(args[++i])); break;
                case "-s": builder.sortBy(args[++i]); break;
                case "-o": builder.outputFormat(args[++i]); break;
                case "-f": builder.outputFile(Paths.get(args[++i].replace("\\", "/"))); break;
                case "--gui": break;
                case "--no-gitignore": builder.useGitIgnore(false); break;
                case "--default-input": builder.defaultInputPath(Paths.get(args[++i].replace("\\", "/"))); break;
                case "--default-output": builder.defaultOutputPath(Paths.get(args[++i].replace("\\", "/"))); break;
                default: System.err.println("Unknown option: " + args[i]); break;
            }
        }
    }
}

package com.codestats.lib.codestats.config;

import java.io.IOException;
import java.nio.file.*;
import java.util.ArrayList;
import java.util.List;
import java.util.stream.Collectors;

public class GitIgnoreLoader {
    public static List<PathMatcher> load(Path rootDir) {
        List<PathMatcher> matchers = new ArrayList<>();
        Path gitignore = rootDir.resolve(".gitignore");
        if (Files.exists(gitignore)) {
            try (java.util.stream.Stream<String> lines = Files.lines(gitignore)) {
                matchers = lines.filter(l -> !l.trim().isEmpty() && !l.startsWith("#"))
                        .map(pattern -> {
                            if (pattern.startsWith("/")) pattern = pattern.substring(1);
                            if (pattern.endsWith("/")) pattern += "**";
                            return FileSystems.getDefault().getPathMatcher("glob:" + pattern);
                        })
                        .collect(Collectors.toList());
            } catch (IOException ignored) {}
        }
        return matchers;
    }
}

package com.codestats.lib.codestats.core;

import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.CopyOnWriteArrayList;
import java.util.concurrent.atomic.AtomicLong;
import java.util.function.BiConsumer;

public class ProgressMonitor {
    private final AtomicLong processed = new AtomicLong(0);
    private final long total;
    private volatile long lastPrint = System.currentTimeMillis();
    private final List<ProgressListener> listeners = new ArrayList<>();

    // 新增：携带文件名的进度监听器（线程安全列表）
    private final List<BiConsumer<Long, Path>> fileProgressListeners = new CopyOnWriteArrayList<>();

    public ProgressMonitor(long total) {
        this.total = total;
    }

    public void addProgressListener(ProgressListener listener) {
        listeners.add(listener);
    }

    /**
     * 新增：注册携带当前文件路径的进度监听器
     */
    public void addFileProgressListener(BiConsumer<Long, Path> listener) {
        fileProgressListeners.add(listener);
    }

    public void increment() {
        long now = processed.incrementAndGet();
        notifyListeners(now, null);
    }

    /**
     * 新增：进度+1，并携带当前处理的文件路径
     */
    public void incrementWithFile(Path currentFile) {
        long now = processed.incrementAndGet();
        notifyListeners(now, currentFile);
    }

    private void notifyListeners(long now, Path currentFile) {
        // 通知原有监听器（仅计数）
        for (ProgressListener listener : listeners) {
            listener.onProgress(now, total);
        }
        // 通知携带文件路径的监听器
        for (BiConsumer<Long, Path> listener : fileProgressListeners) {
            listener.accept(now, currentFile);
        }
        // 控制台输出（保留原有逻辑）
        if (System.currentTimeMillis() - lastPrint > 2000 || now == total) {
            String extra = currentFile != null ? " (" + currentFile.getFileName() + ")" : "";
            System.out.printf("\rProgress: %d / %d (%.1f%%)%s", now, total, 100.0 * now / total, extra);
            lastPrint = System.currentTimeMillis();
            if (now == total) System.out.println();
        }
    }
}

package com.codestats.lib.codestats.filter;
import java.nio.file.Path;
import java.util.List;
import java.util.stream.Collectors;

public class ExtensionFilter implements FileFilter {
    private final List<String> allowed;
    public ExtensionFilter(List<String> exts) { this.allowed = exts.stream().map(String::toLowerCase).collect(Collectors.toList()); }
    @Override
    public boolean accept(Path path) {
        String name = path.toString();
        int dot = name.lastIndexOf('.');
        if (dot == -1) return false;
        String ext = name.substring(dot + 1).toLowerCase();
        return allowed.contains(ext);
    }
}

package com.codestats.lib.codestats.gui;

import com.codestats.lib.codestats.core.FileInfo;
import com.codestats.lib.codestats.core.ProgressMonitor;
import com.codestats.lib.codestats.core.StatisticsResult;
import com.codestats.lib.codestats.filter.CompositeFilter;
import com.codestats.lib.codestats.filter.ExcludePathFilter;
import com.codestats.lib.codestats.filter.ExtensionFilter;
import com.codestats.lib.codestats.pipeline.CollectStage;
import com.codestats.lib.codestats.pipeline.MergeFilesStage;
import com.codestats.lib.codestats.pipeline.ParallelAnalyzeStage;
import com.codestats.lib.codestats.sorter.SorterRegistry;

import javax.swing.*;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.time.Duration;
import java.util.Arrays;
import java.util.List;
import java.util.concurrent.ExecutionException;

public class CodeStatsController {
    private final CodeStatsGUIPanel panel;

    public CodeStatsController(CodeStatsGUIPanel panel) {
        this.panel = panel;
    }

    public void startAnalysis() {
        panel.setStartEnabled(false);
        panel.clearResults();
        panel.resetProgress();

        // 读取用户配置
        String rootDir = panel.getRootDir();
        List<String> extensions = Arrays.asList(panel.getExtensionsText());
        String text = panel.getExcludePathsText();
        List<String> excludePaths;
        if (text == null || text.trim().length() == 0) {
            List<String> strings = new java.util.ArrayList<>();
            excludePaths = strings;
        } else {
            excludePaths = Arrays.asList(panel.getExcludePathsText());
        }
        boolean excludeComments = panel.isExcludeComments();
        int threads = panel.getThreadCount();
        String sortBy = panel.getSortBy();

        // 构建过滤器
        CompositeFilter filterChain = new CompositeFilter();
        if (!extensions.isEmpty()) filterChain.add(new ExtensionFilter(extensions));
        if (!excludePaths.isEmpty()) filterChain.add(new ExcludePathFilter(excludePaths));

        // 后台任务
        new SwingWorker<StatisticsResult, Void>() {
            private long startTime;
            private int directoryCount;
            private int totalFiles;

            @Override
            protected StatisticsResult doInBackground() throws Exception {
                // 1. 收集文件
                CollectStage collectStage = new CollectStage(filterChain);
                List<Path> files = collectStage.process(Paths.get(rootDir));
                if (files.isEmpty()) {
                    throw new Exception("No matching files found.");
                }

                // ★ 新增：合并文件输出（在收集完成后立即执行）
                if (panel.isMergeEnabled()) {
                    String mergePathStr = panel.getMergeOutputFile();
                    if (mergePathStr != null && !mergePathStr.trim().isEmpty()) {
                        Path mergePath = Paths.get(mergePathStr.trim());
                        try {
                            new MergeFilesStage(mergePath).process(files);
                        } catch (Exception e) {
                            System.err.println("Merge failed: " + e.getMessage());
                            // 不阻断分析，仅提示
                        }
                    }
                }

                // 统计目录数
                directoryCount = (int) files.stream().map(Path::getParent).distinct().count();
                totalFiles = files.size();
                SwingUtilities.invokeLater(() -> {
                    panel.updateTotalFiles(totalFiles);
                    panel.updateDirectoryCount(directoryCount);
                });

                // 2. 进度监控（带文件名）
                ProgressMonitor monitor = new ProgressMonitor(totalFiles);
                monitor.addFileProgressListener((processed, currentPath) -> {
                    long elapsed = System.currentTimeMillis() - startTime;
                    SwingUtilities.invokeLater(() -> {
                        int percent = (int) (processed * 100 / totalFiles);
                        String elapsedStr = formatDuration(elapsed);
                        String etaStr = "--:--";
                        if (processed > 0 && processed < totalFiles) {
                            long etaMs = (elapsed * (totalFiles - processed)) / processed;
                            etaStr = formatDuration(etaMs);
                        }
                        panel.updateProgress(percent,
                                currentPath != null ? currentPath.getFileName().toString() : "-",
                                elapsedStr, etaStr);
                    });
                });

                startTime = System.currentTimeMillis();

                // 3. 并行分析
                ParallelAnalyzeStage analyzeStage = new ParallelAnalyzeStage(threads, excludeComments, monitor);
                List<FileInfo> fileInfos = analyzeStage.process(files);

                // 4. 排序与聚合
                fileInfos.sort(SorterRegistry.getComparator(sortBy));
                StatisticsResult result = new StatisticsResult();
                for (FileInfo info : fileInfos) {
                    result.addFileInfo(info);
                }
                result.setDurationMs(System.currentTimeMillis() - startTime);
                return result;
            }

            @Override
            protected void done() {
                try {
                    StatisticsResult result = get();
                    // 更新表格和摘要
                    panel.getTableModel().setFileInfos(result.getFileInfos(), result);

                    StringBuilder sb = new StringBuilder();
                    sb.append("========== Summary ==========\n");
                    sb.append("Total files: ").append(result.getFileInfos().size()).append("\n");
                    sb.append("Total lines: ").append(result.getTotalLines()).append("\n");
                    sb.append("  Code lines: ").append(result.getTotalCodeLines()).append("\n");
                    sb.append("  Comment lines: ").append(result.getTotalCommentLines()).append("\n");
                    sb.append("  Blank lines: ").append(result.getTotalBlankLines()).append("\n");
                    sb.append("Time: ").append(result.getDurationMs()).append(" ms\n\n");
                    sb.append("---------- By Language ----------\n");
                    result.getLanguageSummary().forEach((lang, sum) -> {
                        sb.append(String.format("%-12s : files=%d, lines=%d, code=%d, comment=%d, blank=%d\n",
                                lang, sum.getFiles(), sum.getLines(), sum.getCode(),
                                sum.getComments(), sum.getBlanks()));
                    });
                    panel.setSummaryText(sb.toString());
                    panel.setProgressDone();
                } catch (InterruptedException | ExecutionException e) {
                    JOptionPane.showMessageDialog(panel,
                            "Analysis failed: " + e.getMessage(), "Error", JOptionPane.ERROR_MESSAGE);
                } finally {
                    panel.setStartEnabled(true);
                }
            }
        }.execute();
    }

    private String formatDuration(long millis) {
        Duration d = Duration.ofMillis(millis);
        String format = String.format("%02d:%02d", d.toMinutesPart(), d.toSecondsPart());
        return format;
    }
}

package com.codestats.lib.codestats.language;

import java.io.BufferedReader;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.HashMap;
import java.util.Map;

public class LanguageDetector {
    private static final Map<String, Language> EXTENSION_MAP = new HashMap<>();
    static {
        for (Language lang : Language.values()) {
            LanguageSpec spec = LanguageSpec.forLanguage(lang);
            for (String ext : spec.getExtensions()) {
                EXTENSION_MAP.put(ext.toLowerCase(), lang);
            }
        }
        // 手动补充一些特殊映射
        EXTENSION_MAP.put("kt", Language.KOTLIN);
        EXTENSION_MAP.put("kts", Language.KOTLIN);
        EXTENSION_MAP.put("py", Language.PYTHON);
        EXTENSION_MAP.put("js", Language.JAVASCRIPT);
        EXTENSION_MAP.put("ts", Language.TYPESCRIPT);
        EXTENSION_MAP.put("go", Language.GO);
        EXTENSION_MAP.put("rs", Language.RUST);
    }

    public Language detect(Path file) {
        String fileName = file.getFileName().toString();
        int dot = fileName.lastIndexOf('.');
        if (dot > 0) {
            String ext = fileName.substring(dot + 1).toLowerCase();
            Language lang = EXTENSION_MAP.get(ext);
            if (lang != null) return lang;
        }
        // 检测 shebang
        try (BufferedReader reader = Files.newBufferedReader(file)) {
            String firstLine = reader.readLine();
            if (firstLine != null && firstLine.startsWith("#!")) {
                if (firstLine.contains("python")) return Language.PYTHON;
                if (firstLine.contains("node")) return Language.JAVASCRIPT;
                if (firstLine.contains("ruby")) return Language.RUBY;
                if (firstLine.contains("perl")) return Language.PERL;
                if (firstLine.contains("bash") || firstLine.contains("sh")) return Language.SHELL;
            }
        } catch (Exception ignored) { }
        return Language.UNKNOWN;
    }
}

package com.codestats.lib.codestats.language;

public class LanguageSpec {
    private final Language language;
    private final String lineComment;     // 单行注释起始符，如 //
    private final String blockStart;      // 块注释起始符，如 /*
    private final String blockEnd;        // 块注释结束符，如 */
    private final String[] extensions;    // 关联的扩展名

    public LanguageSpec(Language language, String lineComment, String blockStart, String blockEnd, String[] extensions) {
        this.language = language;
        this.lineComment = lineComment;
        this.blockStart = blockStart;
        this.blockEnd = blockEnd;
        this.extensions = extensions;
    }

    public Language getLanguage() { return language; }
    public String getLineComment() { return lineComment; }
    public String getBlockStart() { return blockStart; }
    public String getBlockEnd() { return blockEnd; }
    public String[] getExtensions() { return extensions; }

    // 工厂方法：内置常见语言
    public static LanguageSpec forLanguage(Language lang) {
        switch (lang) {
            case JAVA: return new LanguageSpec(lang, "//", "/*", "*/", new String[]{"java"});
            case KOTLIN: return new LanguageSpec(lang, "//", "/*", "*/", new String[]{"kt", "kts"});
            case PYTHON: return new LanguageSpec(lang, "#", null, null, new String[]{"py"});
            case JAVASCRIPT: return new LanguageSpec(lang, "//", "/*", "*/", new String[]{"js", "mjs", "cjs"});
            case TYPESCRIPT: return new LanguageSpec(lang, "//", "/*", "*/", new String[]{"ts", "tsx"});
            case GO: return new LanguageSpec(lang, "//", "/*", "*/", new String[]{"go"});
            case RUST: return new LanguageSpec(lang, "//", "/*", "*/", new String[]{"rs"});
            case C: return new LanguageSpec(lang, "//", "/*", "*/", new String[]{"c", "h"});
            case CPP: return new LanguageSpec(lang, "//", "/*", "*/", new String[]{"cpp", "hpp", "cc", "cxx"});
            case CSHARP: return new LanguageSpec(lang, "//", "/*", "*/", new String[]{"cs"});
            case SWIFT: return new LanguageSpec(lang, "//", "/*", "*/", new String[]{"swift"});
            case PHP: return new LanguageSpec(lang, "//", "/*", "*/", new String[]{"php"});
            case RUBY: return new LanguageSpec(lang, "#", "=begin", "=end", new String[]{"rb"});
            case SHELL: return new LanguageSpec(lang, "#", null, null, new String[]{"sh", "bash", "zsh"});
            case SQL: return new LanguageSpec(lang, "--", "/*", "*/", new String[]{"sql"});
            case HTML: return new LanguageSpec(lang, null, "<!--", "-->", new String[]{"html", "htm"});
            case XML: return new LanguageSpec(lang, null, "<!--", "-->", new String[]{"xml"});
            case PROPERTIES: return new LanguageSpec(lang, "#", null, null, new String[]{"properties"});
            case DOCKERFILE: return new LanguageSpec(lang, "#", null, null, new String[]{"dockerfile", "Dockerfile"});
            case MAKEFILE: return new LanguageSpec(lang, "#", null, null, new String[]{"mk", "Makefile"});
            default: return new LanguageSpec(lang, null, null, null, new String[0]);
        }
    }
}

        sb.append(String.format("Time: %d ms\n", result.getDurationMs()));

// ======== D:\work\space\CodeStats\src\main\java\com\demo\lib\codestats\pipeline\AnalyzeStage.java ========
package com.codestats.lib.codestats.pipeline;

import com.codestats.lib.codestats.core.FileInfo;
import com.codestats.lib.codestats.language.Language;
import com.codestats.lib.codestats.language.LanguageDetector;
import com.codestats.lib.codestats.language.LanguageSpec;
import java.io.BufferedReader;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.regex.Pattern;

public class AnalyzeStage implements Stage<Path, FileInfo> {
    private final LanguageDetector detector = new LanguageDetector();
    private final boolean excludeComments;
    private static final Pattern COMPLEXITY_KEYWORDS = Pattern.compile(
            "\\b(if|else|for|while|do|switch|case|catch|\\?\\?)\\b");

    public AnalyzeStage(boolean excludeComments) {
        this.excludeComments = excludeComments;
    }

    @Override
    public FileInfo process(Path file) {
        FileInfo info = new FileInfo(file);
        Language lang = detector.detect(file);
        info.setLanguage(lang);
        LanguageSpec spec = LanguageSpec.forLanguage(lang);
        int complexity = 0;

        if (spec.getLineComment() == null && spec.getBlockStart() == null) {
            // 无法识别，只统计总行数
            long total = countLines(file);
            info.setLines(total);
            if (excludeComments) info.setCodeLines(total);
            return info;
        }

        try (BufferedReader reader = Files.newBufferedReader(file)) {
            long total = 0, code = 0, comment = 0, blank = 0;
            boolean inBlock = false;
            String line;
            while ((line = reader.readLine()) != null) {
                total++;
                String trimmed = line.trim();
                if (trimmed.isEmpty()) {
                    blank++;
                    continue;
                }
                // 复杂度统计（仅对code行）
                if (!inBlock && !trimmed.startsWith(spec.getLineComment() != null ? spec.getLineComment() : "")) {
                    complexity += COMPLEXITY_KEYWORDS.matcher(line).results().count();
                }

                if (inBlock) {
                    comment++;
                    if (spec.getBlockEnd() != null && trimmed.contains(spec.getBlockEnd())) {
                        inBlock = false;
                    }
                    continue;
                }
                if (spec.getBlockStart() != null && trimmed.startsWith(spec.getBlockStart())) {
                    comment++;
                    if (spec.getBlockEnd() != null && !trimmed.contains(spec.getBlockEnd())) {
                        inBlock = true;
                    }
                    continue;
                }
                if (spec.getLineComment() != null && trimmed.startsWith(spec.getLineComment())) {
                    comment++;
                    continue;
                }
                code++;
            }
            info.setLines(total);
            info.setCodeLines(code);
            info.setCommentLines(comment);
            info.setBlankLines(blank);
            info.setComplexity(complexity);
        } catch (Exception e) {
            System.err.println("Error reading " + file + ": " + e.getMessage());
        }
        return info;
    }

    private long countLines(Path file) {
        try (BufferedReader r = Files.newBufferedReader(file)) {
            return r.lines().count();
        } catch (Exception e) {
            return 0;
        }
    }
}

    }

import java.io.IOException;
import java.io.OutputStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.StandardOpenOption;
import java.util.List;

/**
 * 将所有收集到的文件内容合并输出到一个指定文件。
 * 如果目标文件已存在，则先删除再创建。
 */
public class MergeFilesStage implements Stage<List<Path>, Void> {

    private final Path outputFile;

    /**
     * @param outputFile 合并后的输出文件路径（包含文件名）
     */
    public MergeFilesStage(Path outputFile) {
        this.outputFile = outputFile;
    }

    @Override
    public Void process(List<Path> files) throws IOException {
        if (files == null || files.isEmpty()) {
            System.out.println("No files to merge.");
            return null;
        }

        // 如果目标文件已存在，先删除
        Files.deleteIfExists(outputFile);

        // 确保父目录存在
        Path parent = outputFile.getParent();
        if (parent != null && !Files.exists(parent)) {
            Files.createDirectories(parent);
        }

        // 使用缓冲流写入，避免占用过多内存
        try (OutputStream out = Files.newOutputStream(outputFile, StandardOpenOption.CREATE_NEW)) {
            for (Path file : files) {
                if (!Files.isReadable(file)) {
                    System.err.println("Skipping unreadable file: " + file);
                    continue;
                }
                // 写入文件分隔标识（可选）
                String header = "\n// ======== " + file + " ========\n";
                out.write(header.getBytes());
                Files.copy(file, out);
                out.write('\n');
            }
        }

        System.out.println("Merged " + files.size() + " files into: " + outputFile);
        return null;
    }
}

package com.codestats.lib.codestats.pipeline;

import com.codestats.lib.codestats.core.FileInfo;
import com.codestats.lib.codestats.core.ProgressMonitor;

import java.nio.file.Path;
import java.util.List;
import java.util.concurrent.*;
import java.util.stream.Collectors;

public class ParallelAnalyzeStage implements Stage<List<Path>, List<FileInfo>> {
    private final ExecutorService executor;
    private final AnalyzeStage analyzeStage;
    private final ProgressMonitor monitor;

    public ParallelAnalyzeStage(int threads, boolean excludeComments, ProgressMonitor monitor) {
        this.executor = Executors.newFixedThreadPool(threads);
        this.analyzeStage = new AnalyzeStage(excludeComments);
        this.monitor = monitor;
    }

    @Override
    public List<FileInfo> process(List<Path> paths) {
        List<CompletableFuture<FileInfo>> futures = paths.stream()
                .map(path -> CompletableFuture.supplyAsync(() -> {
                    FileInfo info = analyzeStage.process(path);
                    // 使用新的携带文件路径的通知方法（若monitor非空）
                    if (monitor != null) {
                        monitor.incrementWithFile(path);
                    }
                    return info;
                }, executor))
                .collect(Collectors.toList());
        List<FileInfo> result = futures.stream().map(CompletableFuture::join).collect(Collectors.toList());
        executor.shutdown();
        return result;
    }
}

package com.codestats.lib.codestats.pipeline;

import com.codestats.lib.codestats.config.Config;
import com.codestats.lib.codestats.core.ProgressMonitor;
import com.codestats.lib.codestats.filter.CompositeFilter;
import com.codestats.lib.codestats.filter.ExcludePathFilter;
import com.codestats.lib.codestats.filter.ExtensionFilter;
import com.codestats.lib.codestats.filter.GitIgnoreFilter;

import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;

public class Pipeline<T, R> {
    private final List<Stage> stages = new ArrayList<>();

    // 用于向 OutputStage 传递管道开始时间（同一线程顺序执行，安全）
    static final ThreadLocal<Long> START_TIME_HOLDER = new ThreadLocal<>();

    public Pipeline<T, R> addStage(Stage<?, ?> stage) {
        stages.add(stage);
        return this;
    }

    @SuppressWarnings("unchecked")
    public R execute(T input) {
        // 记录管道开始时间
        START_TIME_HOLDER.set(System.currentTimeMillis());
        try {
            Object current = input;
            for (Stage stage : stages) {
                current = stage.process(current);
            }
            return (R) current;
        } catch (Exception e) {
            throw new RuntimeException("Pipeline execution failed", e);
        } finally {
            START_TIME_HOLDER.remove();
        }
    }

    public static Pipeline<Path, Void> createDefault(Config config) {
        // 构建过滤器链
        CompositeFilter filterChain = new CompositeFilter();
        if (config.getExtensions() != null && !config.getExtensions().isEmpty()) {
            filterChain.add(new ExtensionFilter(config.getExtensions()));
        }
        if (config.getExcludePaths() != null && !config.getExcludePaths().isEmpty()) {
            filterChain.add(new ExcludePathFilter(config.getExcludePaths()));
        }
        if (config.isUseGitIgnore()) {
            filterChain.add(new GitIgnoreFilter(config.getRootDir()));
        }

        Pipeline<Path, Void> pipeline = new Pipeline<>();
        pipeline.addStage(new CollectStage(filterChain));
        ProgressMonitor monitor = null; // 控制台模式不绑定进度条
        pipeline.addStage(new ParallelAnalyzeStage(config.getThreadCount(), config.isExcludeComments(), monitor));
        pipeline.addStage(new AggregateStage(config.getSortBy()));
        try {
            pipeline.addStage(new OutputStage(config));
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
        return pipeline;
    }
}

package com.codestats.lib.codestats.statistics;


import com.codestats.lib.codestats.core.FileInfo;

import java.io.BufferedReader;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;



// 行数统计（支持排除空行/注释）
public class LineCountStrategy implements StatisticStrategy {
    private final boolean excludeCommentsAndBlanks;

    public LineCountStrategy(boolean excludeCommentsAndBlanks) {
        this.excludeCommentsAndBlanks = excludeCommentsAndBlanks;
    }

    @Override
    public void analyze(Path file, FileInfo info) {
        try (BufferedReader reader = Files.newBufferedReader(file)) {
            String line;
            long total = 0, code = 0, comment = 0, blank = 0;
            boolean inBlockComment = false;
            while ((line = reader.readLine()) != null) {
                total++;
                String trimmed = line.trim();
                if (trimmed.isEmpty()) {
                    blank++;
                    continue;
                }
                // 简化的注释检测（实际项目需要更严谨）
                if (excludeCommentsAndBlanks) {
                    if (trimmed.startsWith("//")) {
                        comment++;
                    } else if (trimmed.startsWith("/*")) {
                        comment++;
                        inBlockComment = true;
                    } else if (trimmed.endsWith("*/") && inBlockComment) {
                        comment++;
                        inBlockComment = false;
                    } else if (inBlockComment) {
                        comment++;
                    } else {
                        code++;
                    }
                } else {
                    code++;
                }
            }
            info.setLines(total);
            info.setCodeLines(code);
            info.setCommentLines(comment);
            info.setBlankLines(blank);
        } catch (IOException e) {
            System.err.println("Error reading " + file + ": " + e.getMessage());
        }
    }
}

package com.codestats.lib.codestats.util;

import java.io.IOException;
import java.nio.file.FileSystems;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.PathMatcher;
import java.util.List;
import java.util.stream.Collectors;

public class GitIgnoreLoader {
    public static List<PathMatcher> load(Path rootDir) {
        Path gitignore = rootDir.resolve(".gitignore");
        if (!Files.exists(gitignore)) {
            List<PathMatcher> pathMatchers = new java.util.ArrayList<>();
            return pathMatchers;
        }
        try (java.util.stream.Stream<String> lines = Files.lines(gitignore)) {
            return lines.filter(l -> !l.trim().isEmpty() && !l.startsWith("#"))
                    .map(pattern -> {
                        if (pattern.startsWith("/")) pattern = pattern.substring(1);
                        if (pattern.endsWith("/")) pattern += "**";
                        return FileSystems.getDefault().getPathMatcher("glob:" + pattern);
                    })
                    .collect(Collectors.toList());
        } catch (IOException e) {
            List<PathMatcher> pathMatchers = new java.util.ArrayList<>();
            return pathMatchers;
        }
    }
}

package com.framework.jdbc;

import java.sql.*;
import java.util.ArrayList;
import java.util.List;

public class JdbcTemplate {

    /**
     * 执行查询，支持超时设置
     * @param sql SQL语句
     * @param timeoutSeconds 超时秒数，<=0 则不设置超时
     * @param rowMapper 行映射器
     * @param args 参数
     * @return 结果列表
     */
    public <T> List<T> query(String sql, int timeoutSeconds, RowMapper<T> rowMapper, Object... args) {
        List<T> results = new ArrayList<>();
        try (Connection conn = DataSourceUtils.getConnection();
             PreparedStatement ps = conn.prepareStatement(sql)) {
            // 设置超时
            if (timeoutSeconds > 0) {
                ps.setQueryTimeout(timeoutSeconds);
            }
            for (int i = 0; i < args.length; i++) {
                ps.setObject(i + 1, args[i]);
            }
            try (ResultSet rs = ps.executeQuery()) {
                while (rs.next()) {
                    results.add(rowMapper.mapRow(rs));
                }
            }
        } catch (SQLException e) {
            throw new RuntimeException("Query failed: " + sql, e);
        }
        return results;
    }

    /**
     * 查询单条记录，支持超时
     */
    public <T> T queryOne(String sql, int timeoutSeconds, RowMapper<T> rowMapper, Object... args) {
        List<T> list = query(sql, timeoutSeconds, rowMapper, args);
        return list.isEmpty() ? null : list.get(0);
    }

    @FunctionalInterface
    public interface RowMapper<T> {
        T mapRow(ResultSet rs) throws SQLException;
    }
}

public class SimpleRow implements Row {

    private final List<Object> values;      // 按列顺序存储的值

    public SimpleRow(IDataModel model, List<Object> values) {
        this.model = model;
        this.values = values;
    }

    @Override
    public Object getValue(int columnIndex) {
        if (columnIndex < 0 || columnIndex >= values.size()) {
            throw new IndexOutOfBoundsException("Column index out of range: " + columnIndex);
        }
        return values.get(columnIndex);
    }

    @Override
    public Object getValue(String columnName) {
        int idx = model.getColumnIndex(columnName);
        if (idx == -1) {
            throw new IllegalArgumentException("Column not found: " + columnName);
        }
        return values.get(idx);
    }

    @Override
    public int getColumnCount() {
        return values.size();
    }

    @Override
    public String[] getColumnNames() {
        return model.getColumnNames();
    }

    @Override
    public IDataModel getModel() {
        return model;
    }
}

/**

 */
public class MetadataConverter {

    public static DatabaseInfo toDatabaseInfo(ResultSet rs) throws SQLException {
        // TABLE_CAT 即为 catalog（数据库名）
        String name = rs.getString("TABLE_CAT");
        return new DatabaseInfo(name);
    }

    public static TableInfo toTableInfo(ResultSet rs) throws SQLException {
        String catalog = rs.getString("TABLE_CAT");
        String schema = rs.getString("TABLE_SCHEM");
        String name = rs.getString("TABLE_NAME");
        String type = rs.getString("TABLE_TYPE");
        String remarks = rs.getString("REMARKS");
        return new TableInfo(catalog, schema, name, type, remarks);
    }

    public static ColumnInfo toColumnInfo(ResultSet rs) throws SQLException {
        String columnName = rs.getString("COLUMN_NAME");
        int dataType = rs.getInt("DATA_TYPE");
        String typeName = rs.getString("TYPE_NAME");
        int columnSize = rs.getInt("COLUMN_SIZE");
        int decimalDigits = rs.getInt("DECIMAL_DIGITS");
        boolean nullable = rs.getInt("NULLABLE") == DatabaseMetaData.columnNullable;

        String defaultValue = rs.getString("COLUMN_DEF");
        int ordinalPosition = rs.getInt("ORDINAL_POSITION");
        boolean autoIncrement = "YES".equalsIgnoreCase(rs.getString("IS_AUTOINCREMENT"));
        boolean generated = "YES".equalsIgnoreCase(rs.getString("IS_GENERATEDCOLUMN"));
        return new ColumnInfo(columnName, dataType, typeName, columnSize, decimalDigits,
                nullable, remarks, defaultValue, ordinalPosition, autoIncrement, generated);
    }

package com.framework.mybatis.proxy;

import com.framework.jdbc.JdbcTemplate;
import com.framework.mybatis.annotation.Select;


import java.lang.reflect.InvocationHandler;
import java.lang.reflect.Method;
import java.util.List;

public class MapperProxy implements InvocationHandler {
    private final JdbcTemplate jdbcTemplate;

    public MapperProxy(JdbcTemplate jdbcTemplate) {
        this.jdbcTemplate = jdbcTemplate;
    }

    @Override
    public Object invoke(Object proxy, Method method, Object[] args) throws Throwable {
        if (method.getDeclaringClass() == Object.class) {
            return method.invoke(this, args);
        }
        Select select = method.getAnnotation(Select.class);
        if (select == null) {
            throw new UnsupportedOperationException("Mapper method must have @Select: " + method);
        }
        String sql = select.value();
        int timeout = select.timeout();          // 获取注解中的超时设置
        Object[] paramValues = args == null ? new Object[0] : args;
        Class<?> returnType = method.getReturnType();

        if (returnType == List.class) {
            return jdbcTemplate.query(sql, timeout, rs -> rs.getObject(1).toString(), paramValues);
        } else {
            return jdbcTemplate.queryOne(sql, timeout, rs -> {
                if (returnType == String.class) return rs.getString(1);
                else if (returnType == Integer.class) return rs.getInt(1);
                else return rs.getObject(1).toString();
            }, paramValues);
        }
    }
}

package com.framework.pool;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.SQLException;
import java.sql.Statement;
import java.util.concurrent.ArrayBlockingQueue;
import java.util.concurrent.BlockingQueue;
import java.util.concurrent.Semaphore;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicInteger;

public class SimpleDataSource {
    private final String url;
    private final String username;
    private final String password;
    private final int maxActive;
    private final int maxIdle;
    private final int minIdle;
    private final long maxWaitMillis;
    private final String validationQuery;

    private final BlockingQueue<Connection> idleConnections;
    private final Semaphore semaphore;          // 控制最大活跃连接数
    private final AtomicInteger activeCount = new AtomicInteger(0);

    private volatile boolean closed = false;

    private SimpleDataSource(Builder builder) {
        this.url = builder.url;
        this.username = builder.username;
        this.password = builder.password;
        this.maxActive = builder.maxActive;
        this.maxIdle = builder.maxIdle;
        this.minIdle = builder.minIdle;
        this.maxWaitMillis = builder.maxWaitMillis;
        this.validationQuery = builder.validationQuery;
        this.semaphore = new Semaphore(maxActive);

        // 初始化 minIdle 个连接
        idleConnections = new ArrayBlockingQueue<>(maxIdle);
        for (int i = 0; i < minIdle; i++) {
            try {
                Connection conn = createNewConnection();
                idleConnections.offer(conn);
            } catch (SQLException e) {
                throw new RuntimeException("Failed to create initial connection", e);
            }
        }
    }

    private Connection createNewConnection() throws SQLException {
        return DriverManager.getConnection(url, username, password);
    }

    public Connection getConnection() throws SQLException {
        if (closed) {
            throw new SQLException("DataSource is closed");
        }
        // 先尝试获取许可证（控制最大活跃数）
        boolean acquired;
        try {
            if (maxWaitMillis > 0) {
                acquired = semaphore.tryAcquire(maxWaitMillis, TimeUnit.MILLISECONDS);
            } else {
                semaphore.acquire();
                acquired = true;
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new SQLException("Interrupted while waiting for connection", e);
        }
        if (!acquired) {
            throw new SQLException("Timeout waiting for connection");
        }

        // 从空闲队列中获取连接
        Connection conn = idleConnections.poll();
        if (conn == null) {
            // 没有空闲连接，创建新连接（但已经通过 semaphore 限制了总数）
            try {
                conn = createNewConnection();
            } catch (SQLException e) {
                semaphore.release();
                throw e;
            }
        }
        activeCount.incrementAndGet();
        // 验证连接是否有效
        if (!isValid(conn)) {
            try {
                conn.close();
            } catch (SQLException ignored) {}
            // 递归重试获取一个有效连接
            return getConnection();
        }
        return new PooledConnection(conn, this);
    }

    void returnConnection(Connection conn) {
        if (closed) {
            try { conn.close(); } catch (SQLException ignored) {}
            return;
        }
        activeCount.decrementAndGet();
        // 如果空闲连接数未超过 maxIdle，则放回队列，否则直接关闭
        if (idleConnections.size() < maxIdle) {
            if (isValid(conn)) {
                idleConnections.offer(conn);
            } else {
                try { conn.close(); } catch (SQLException ignored) {}
            }
        } else {
            try { conn.close(); } catch (SQLException ignored) {}
        }
        semaphore.release();
    }

    private boolean isValid(Connection conn) {
        if (validationQuery == null || validationQuery.isEmpty()) {
            return true;
        }
        try (Statement stmt = conn.createStatement()) {
            stmt.execute(validationQuery);
            return true;
        } catch (SQLException e) {
            return false;
        }
    }

    public void close() {
        closed = true;
        for (Connection conn : idleConnections) {
            try { conn.close(); } catch (SQLException ignored) {}
        }
        idleConnections.clear();
    }

    // Builder 模式
    public static class Builder {
        private String url;
        private String username;
        private String password;
        private int maxActive = 10;
        private int maxIdle = 5;
        private int minIdle = 2;
        private long maxWaitMillis = 30000;
        private String validationQuery = "SELECT 1";

        public Builder url(String url) { this.url = url; return this; }
        public Builder username(String username) { this.username = username; return this; }
        public Builder password(String password) { this.password = password; return this; }
        public Builder maxActive(int maxActive) { this.maxActive = maxActive; return this; }
        public Builder maxIdle(int maxIdle) { this.maxIdle = maxIdle; return this; }
        public Builder minIdle(int minIdle) { this.minIdle = minIdle; return this; }
        public Builder maxWaitMillis(long maxWaitMillis) { this.maxWaitMillis = maxWaitMillis; return this; }
        public Builder validationQuery(String validationQuery) { this.validationQuery = validationQuery; return this; }

        public SimpleDataSource build() {
            if (url == null) throw new IllegalArgumentException("url is required");
            return new SimpleDataSource(this);
        }
    }
}

package com.framework.spring.ioc;


import com.framework.jdbc.JdbcTemplate;
import com.framework.mybatis.proxy.MapperProxy;
import com.framework.spring.annotation.*;
import com.framework.tomcat.util.PackageScanner;

import java.lang.annotation.Annotation;
import java.lang.reflect.Field;
import java.lang.reflect.Proxy;
import java.util.*;

public class SimpleApplicationContext {

    private final Map<Class<?>, Object> beans = new HashMap<>();
    private final List<Object> allBeans = new ArrayList<>();
    private JdbcTemplate jdbcTemplate;

    public void scanAndInit(String packagesToScan) throws Exception {
        jdbcTemplate = new JdbcTemplate();

        String[] packages = packagesToScan.split("[;, ]+");
        Set<Class<?>> allClasses = PackageScanner.getClasses(packages);

        // 1. 创建普通 Bean
        for (Class<?> clazz : allClasses) {
            if (clazz.isInterface()) continue;
            if (clazz.isAnnotationPresent(Service.class) ||
                    clazz.isAnnotationPresent(Repository.class) ||
                    clazz.isAnnotationPresent(Controller.class) ||
                    clazz.isAnnotationPresent(Component.class)) {   // 新增此行
                Object instance = clazz.getDeclaredConstructor().newInstance();
                beans.put(clazz, instance);
                allBeans.add(instance);
                System.out.println("IoC created bean: " + clazz.getName());
            }
        }

        // 2. 生成 Mapper 接口的动态代理
        for (Class<?> clazz : allClasses) {
            if (clazz.isInterface() && clazz.isAnnotationPresent(Mapper.class)) {
                MapperProxy handler = new MapperProxy(jdbcTemplate);
                Object proxy = Proxy.newProxyInstance(clazz.getClassLoader(), new Class[]{clazz}, handler);
                beans.put(clazz, proxy);
                allBeans.add(proxy);
                System.out.println("IoC created mapper proxy: " + clazz.getName());
            }
        }

        // 3. 注入 @Autowired
        for (Object bean : allBeans) {
            for (Field field : bean.getClass().getDeclaredFields()) {
                if (field.isAnnotationPresent(Autowired.class)) {
                    field.setAccessible(true);
                    Class<?> fieldType = field.getType();
                    Object dependency = beans.get(fieldType);
                    if (dependency == null) {
                        dependency = findImplementation(fieldType);
                    }
                    if (dependency == null) {
                        throw new RuntimeException("No bean found for type: " + fieldType.getName());
                    }
                    field.set(bean, dependency);
                    System.out.println("Autowired: " + bean.getClass().getSimpleName() + " -> " + fieldType.getSimpleName());
                }
            }
        }
    }

    private Object findImplementation(Class<?> interfaceType) {
        for (Object bean : allBeans) {
            if (interfaceType.isAssignableFrom(bean.getClass())) {
                return bean;
            }
        }
        return null;
    }

    public List<Object> getBeansWithAnnotation(Class<? extends Annotation> annotationClass) {
        List<Object> result = new ArrayList<>();
        for (Object bean : allBeans) {
            if (bean.getClass().isAnnotationPresent(annotationClass)) {
                result.add(bean);
            }
        }
        return result;
    }

    public <T> T getBean(Class<T> type) {
        return (T) beans.get(type);
    }
}

package com.framework.spring.mvc;

import com.framework.spring.annotation.Controller;
import com.framework.spring.annotation.RequestMapping;
import com.framework.spring.config.ConfigLoader;
import com.framework.spring.ioc.SimpleApplicationContext;
import com.framework.tomcat.connector.Request;
import com.framework.tomcat.connector.Response;
import com.framework.tomcat.servlet.HttpServlet;

import java.io.File;
import java.io.FileInputStream;
import java.io.IOException;
import java.io.OutputStream;
import java.lang.reflect.Method;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class DispatcherServlet extends HttpServlet {

    private SimpleApplicationContext applicationContext;
    private final Map<String, Handler> handlerMappings = new HashMap<>();

    // 静态资源存放的根目录（相对于当前工作目录，一般为项目根目录）
    private static final String STATIC_BASE = "webapps/demo";

    @Override
    public void init() throws Exception {
        System.out.println(">>> DispatcherServlet init: scanning and creating IoC container...");
        applicationContext = new SimpleApplicationContext();
        String packages = ConfigLoader.getProperty("scan.packages",
                "com.codestats.*.business");
        applicationContext.scanAndInit(packages);
        List<Object> controllers = applicationContext.getBeansWithAnnotation(Controller.class);
        System.out.println(">>> Controllers found: " + controllers.size());

        for (Object controller : controllers) {
            Class<?> clazz = controller.getClass();
            String basePath = "";
            if (clazz.isAnnotationPresent(RequestMapping.class)) {
                basePath = clazz.getAnnotation(RequestMapping.class).value();
            }
            for (Method method : clazz.getDeclaredMethods()) {
                if (method.isAnnotationPresent(RequestMapping.class)) {
                    String methodPath = method.getAnnotation(RequestMapping.class).value();
                    String fullPath = (basePath + methodPath).replaceAll("//", "/");
                    if (!fullPath.startsWith("/")) fullPath = "/" + fullPath;
                    handlerMappings.put(fullPath, new Handler(controller, method));
                    System.out.println("Mapped " + fullPath + " -> " + clazz.getSimpleName() + "." + method.getName());
                }
            }
        }

        System.out.println(">>> DispatcherServlet ready, mapped " + handlerMappings.size() + " handlers");
    }

    @Override
    protected void doGet(Request req, Response res) throws Exception {
        processRequest(req, res);
    }

    @Override
    protected void doPost(Request req, Response res) throws Exception {
        processRequest(req, res);
    }

    private void processRequest(Request req, Response res) throws Exception {
        RequestContextHolder.setRequest(req);
        try {
            String uri = req.getUri();
            System.out.println(">>> Request URI: " + uri);

            // 1. 尝试匹配动态 Handler
            Handler handler = handlerMappings.get(uri);
            if (handler != null) {
                try {
                    Object result = handler.method.invoke(handler.controller);
                    if (result != null) {
                        res.write(result.toString());
                    } else {
                        res.write("OK");
                    }
                    res.send();
                } catch (Exception e) {
                    e.printStackTrace();
                    res.setStatus(500, "Internal Server Error");
                    res.write("<h1>500 - " + getRootCauseMessage(e) + "</h1>");
                    res.send();
                }
                return;
            }

            // 2. 尝试提供静态文件
            if (serveStaticFile(uri, res)) {
                return;
            }

            // 3. 都不匹配，返回 404
            res.setStatus(404, "Not Found");
            res.write("<h1>404 - No handler for " + uri + "</h1>");
            res.write("<hr/>Available mappings:<br/>");
            for (String key : handlerMappings.keySet()) {
                res.write(key + "<br/>");
            }
            res.send();
        } finally {
            RequestContextHolder.clear();
        }
    }

    /**
     * 尝试从静态目录中读取文件并写入响应。
     * @return true 表示文件存在并已发送，false 表示文件不存在
     */
    private boolean serveStaticFile(String uri, Response res) {
        // 安全检查，防止路径穿越
        if (uri.contains("..") || uri.contains("//")) {
            return false;
        }
        Path filePath = Paths.get(STATIC_BASE, uri);
        File file = filePath.toFile();

        // ★ 添加调试日志，查看实际查找的绝对路径
        System.out.println(">>> Static file lookup: " + filePath.toAbsolutePath());

        if (!file.exists() || file.isDirectory()) {
            System.out.println(">>> Static file NOT FOUND: " + filePath.toAbsolutePath());
            return false;
        }

        String contentType = getContentType(uri);
        res.setContentType(contentType);
        res.setStatus(200, "OK");

        try (FileInputStream fis = new FileInputStream(file)) {
            res.sendStatic(fis, contentType, file.length());
            System.out.println(">>> Static file served: " + uri);
            return true;
        } catch (IOException e) {
            e.printStackTrace();
            return false;
        }
    }

    private String getContentType(String uri) {
        if (uri.endsWith(".html") || uri.endsWith(".htm")) return "text/html; charset=utf-8";
        if (uri.endsWith(".css")) return "text/css; charset=utf-8";
        if (uri.endsWith(".js")) return "application/javascript; charset=utf-8";
        if (uri.endsWith(".json")) return "application/json; charset=utf-8";
        if (uri.endsWith(".png")) return "image/png";
        if (uri.endsWith(".jpg") || uri.endsWith(".jpeg")) return "image/jpeg";
        return "application/octet-stream";
    }

    public static String getRootCauseMessage(Throwable t) {
        while (t.getCause() != null && t.getCause() != t) {
            t = t.getCause();
        }
        String msg = t.getMessage();
        return msg != null ? msg : t.getClass().getSimpleName();
    }

    private static class Handler {
        Object controller;
        Method method;
        Handler(Object controller, Method method) {
            this.controller = controller;
            this.method = method;
        }
    }
}

import com.framework.tomcat.core.Lifecycle;
import com.framework.tomcat.core.LifecycleListener;
import com.framework.tomcat.core.LifecycleState;

import java.io.IOException;
import java.net.ServerSocket;
import java.net.Socket;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

public class Connector implements Lifecycle {
    private int port;
    private Service service;
    private ExecutorService threadPool;
    private volatile boolean running;
    private ServerSocket serverSocket;

    public Connector(int port) { this.port = port; }

    public void setService(Service service) { this.service = service; }

    @Override
    public void init() { threadPool = Executors.newCachedThreadPool(); }

    @Override
    public void start() {
        running = true;
        new Thread(() -> {
            try {
                serverSocket = new ServerSocket(port);
                System.out.println("Connector listening on " + port);
                while (running) {
                    Socket socket = serverSocket.accept();
                    threadPool.submit(() -> process(socket));
                }
            } catch (IOException e) { e.printStackTrace(); }
        }).start();
    }

    private void process(Socket socket) {
        try (java.io.InputStream in = socket.getInputStream(); java.io.OutputStream out = socket.getOutputStream()) {
            Request request = new Request(in);
            request.parse();
            Response response = new Response(out);
            Engine engine = service.getEngine();
            if (engine != null) {
                engine.getPipeline().invoke(request, response);
            } else {
                response.setStatus(500, "Internal Server Error");
                response.write("<h1>No Engine</h1>");
                response.send();
            }
        } catch (Exception e) { e.printStackTrace(); }
        finally { try { socket.close(); } catch (IOException e) {} }
    }

    @Override
    public void stop() { running = false; try { if (serverSocket != null) serverSocket.close(); } catch (IOException e) {} }
    @Override
    public void destroy() { stop(); if (threadPool != null) threadPool.shutdown(); }
    @Override public void addLifecycleListener(LifecycleListener listener) {}
    @Override public LifecycleListener[] findLifecycleListeners() { return new LifecycleListener[0]; }
    @Override public void removeLifecycleListener(LifecycleListener listener) {}
    @Override public LifecycleState getState() { return null; }
}

package com.framework.tomcat.connector;

import java.io.BufferedReader;
import java.io.InputStream;
import java.io.InputStreamReader;

public class Request {
    private final InputStream input;
    private String uri;
    private String method;
    private String contextPath = "";
    private String queryString;  // 新增

    public Request(InputStream input) { this.input = input; }

    public void parse() throws Exception {
        BufferedReader reader = new BufferedReader(new InputStreamReader(input));
        String line = reader.readLine();
        if (line != null) {
            String[] parts = line.split(" ");
            if (parts.length >= 2) {
                method = parts[0];
                String fullUri = parts[1];
                // 分离 URI 和查询字符串
                int qIdx = fullUri.indexOf('?');
                if (qIdx >= 0) {
                    queryString = fullUri.substring(qIdx + 1);
                    fullUri = fullUri.substring(0, qIdx);
                }
                // 分离 contextPath 和 servletPath
                int idx = fullUri.indexOf('/', 1);
                if (idx > 0) {
                    contextPath = fullUri.substring(0, idx);
                    uri = fullUri.substring(idx);
                } else {
                    contextPath = "";
                    uri = fullUri;
                }
                if (uri.isEmpty()) uri = "/";
            }
        }
        while ((line = reader.readLine()) != null && !line.isEmpty()) {}
    }

    public String getUri() { return uri; }
    public String getMethod() { return method; }
    public String getContextPath() { return contextPath; }
    public String getQueryString() { return queryString; }  // 新增
}

package com.framework.tomcat.connector;

import java.io.*;
import java.nio.charset.StandardCharsets;

public class Response {
    private final OutputStream output;
    private int statusCode = 200;
    private String statusMessage = "OK";
    private final StringBuilder content = new StringBuilder();
    private String contentType = "text/html; charset=utf-8";  // 默认

    public Response(OutputStream output) { this.output = output; }

    public void setStatus(int code, String message) { this.statusCode = code; this.statusMessage = message; }
    public void write(String text) { content.append(text); }
    public void setContentType(String contentType) { this.contentType = contentType; }
    public OutputStream getRawOutputStream() { return output; }

    // 原有的 send 方法，发送字符内容
    public void send() {
        try {
            String body = content.toString();
            byte[] bytes = body.getBytes(StandardCharsets.UTF_8);
            PrintWriter writer = new PrintWriter(output, true);
            writer.print("HTTP/1.1 " + statusCode + " " + statusMessage + "\r\n");
            writer.print("Content-Type: " + contentType + "\r\n");
            writer.print("Content-Length: " + bytes.length + "\r\n");
            writer.print("\r\n");
            writer.flush();
            output.write(bytes);
            output.flush();
        } catch (IOException e) {
            System.err.println("Response send failed: " + e.getMessage());
        }
    }

    /**
     * 发送静态文件（二进制流）
     */
    public void sendStatic(InputStream inputStream, String contentType, long contentLength) throws IOException {
        this.contentType = contentType;
        this.statusCode = 200;
        this.statusMessage = "OK";

        PrintWriter writer = new PrintWriter(output, true);
        writer.print("HTTP/1.1 200 OK\r\n");
        writer.print("Content-Type: " + contentType + "\r\n");
        writer.print("Content-Length: " + contentLength + "\r\n");
        writer.print("\r\n");
        writer.flush();

        byte[] buffer = new byte[4096];
        int bytesRead;
        while ((bytesRead = inputStream.read(buffer)) != -1) {
            output.write(buffer, 0, bytesRead);
        }
        output.flush();
    }
}

package com.framework.tomcat.container;

import com.framework.tomcat.connector.Request;
import com.framework.tomcat.connector.Response;

import java.util.ArrayList;
import java.util.List;

public class SimplePipeline implements Pipeline {
    private final List<Valve> valves = new ArrayList<>();
    private Valve basic;
    private final Container container;

    public SimplePipeline(Container container) { this.container = container; }

    @Override public Valve getBasic() { return basic; }
    @Override public void setBasic(Valve valve) { this.basic = valve; }
    @Override public void addValve(Valve valve) { valves.add(valve); }
    @Override public Valve[] getValves() { return valves.toArray(new Valve[0]); }
    @Override public void removeValve(Valve valve) { valves.remove(valve); }

    @Override
    public void invoke(Request request, Response response) throws Exception {
        new ValveChainImpl().invokeNext(request, response);
    }

    private class ValveChainImpl implements Valve.ValveChain {
        private int index = 0;

        @Override
        public void invokeNext(Request request, Response response) throws Exception {
            if (index < valves.size()) {
                valves.get(index++).invoke(request, response, this);
            } else if (basic != null) {
                basic.invoke(request, response, this);
            } else {
                // 默认处理：交给容器或子容器
                if (container instanceof Wrapper) {
                    ((StandardWrapper) container).invokeServlet(request, response);
                } else if (container instanceof Context) {
                    String path = request.getUri();
                    String servletName = ((Context) container).findServletMapping(path);
                    if (servletName != null) {
                        Container wrapper = container.findChild(servletName);
                        if (wrapper != null) wrapper.getPipeline().invoke(request, response);
                        else fallback404(response);
                    } else fallback404(response);
                } else if (container instanceof Host) {
                    String contextPath = request.getContextPath();
                    Container context = container.findChild(contextPath);
                    if (context != null) context.getPipeline().invoke(request, response);
                    else fallback404(response);
                } else if (container instanceof Engine) {
                    Container host = container.findChild(((Engine) container).getDefaultHost());
                    if (host != null) host.getPipeline().invoke(request, response);
                    else fallback404(response);
                }
            }
        }

        private void fallback404(Response response) throws Exception {
            response.setStatus(404, "Not Found");
            response.write("<h1>404 - Not Found</h1>");
            response.send();
        }
    }
}

import com.framework.tomcat.core.Lifecycle;
import com.framework.tomcat.core.LifecycleState;

import java.io.File;
import java.net.URL;
import java.util.HashMap;
import java.util.Map;

public class StandardContext extends ContainerBase implements Context {
    private String docBase;
    private ClassLoader parentLoader;
    private WebappClassLoader webappLoader;
    private final Map<String, String> servletMappings = new HashMap<>();
    private final Map<String, Filter> filters = new HashMap<>();
    private String defaultServletName;  // 新增：存储 /* 映射的 servlet 名

    @Override public String getDocBase() { return docBase; }
    @Override public void setDocBase(String docBase) { this.docBase = docBase; }

    public void setParentClassLoader(ClassLoader parent) { this.parentLoader = parent; }

    @Override
    public void init() {
        state = LifecycleState.INITIALIZING;
        fireLifecycleEvent(Lifecycle.INIT_EVENT, null);
        try {
            URL classesUrl = new File(docBase + "/WEB-INF/classes/").toURI().toURL();
            URL libUrl = new File(docBase + "/WEB-INF/lib/").toURI().toURL();
            webappLoader = new WebappClassLoader(parentLoader, new URL[]{classesUrl, libUrl});
        } catch (Exception e) { webappLoader = new WebappClassLoader(parentLoader, new URL[0]); }
        for (Container child : findChildren()) child.init();
        state = LifecycleState.INITIALIZED;
    }

    @Override
    public void start() {
        state = LifecycleState.STARTING;
        fireLifecycleEvent(Lifecycle.START_EVENT, null);
        for (Container child : findChildren()) child.start();
        state = LifecycleState.STARTED;
    }

    @Override
    public void stop() {
        state = LifecycleState.STOPPING;
        fireLifecycleEvent(Lifecycle.STOP_EVENT, null);
        for (Container child : findChildren()) child.stop();
        state = LifecycleState.STOPPED;
    }

    @Override
    public void destroy() {
        state = LifecycleState.DESTROYING;
        fireLifecycleEvent(Lifecycle.DESTROY_EVENT, null);
        for (Container child : findChildren()) child.destroy();
        if (webappLoader != null) try { webappLoader.close(); } catch (Exception e) {}
        state = LifecycleState.DESTROYED;
    }

    @Override
    public void addServletMapping(String pattern, String servletName) {
        servletMappings.put(pattern, servletName);
        if ("/*".equals(pattern)) {
            this.defaultServletName = servletName;
        }
    }

    @Override
    public String findServletMapping(String uri) {
        // 1. 精确匹配
        String servletName = servletMappings.get(uri);
        if (servletName != null) {
            return servletName;
        }
        // 2. 匹配 /*（默认 Servlet）
        if (defaultServletName != null) {
            return defaultServletName;
        }
        return null;
    }

    @Override public void addFilter(Filter filter, String filterName) { filters.put(filterName, filter); }
    @Override public Filter findFilter(String filterName) { return filters.get(filterName); }
    @Override public String[] findFilters() { return filters.keySet().toArray(new String[0]); }
    @Override public ClassLoader getLoader() { return webappLoader; }
}

package com.framework.tomcat.container;

import com.framework.tomcat.core.Lifecycle;
import com.framework.tomcat.core.LifecycleState;
import com.framework.tomcat.filter.ApplicationFilterChain;
import com.framework.tomcat.filter.Filter;
import com.framework.tomcat.servlet.Servlet;
import com.framework.tomcat.connector.Request;
import com.framework.tomcat.connector.Response;


public class StandardWrapper extends ContainerBase implements Wrapper {
    private String servletClass;
    private Servlet servletInstance;

    @Override public String getServletClass() { return servletClass; }
    @Override public void setServletClass(String servletClass) { this.servletClass = servletClass; }

    @Override
    public void init() {
        state = LifecycleState.INITIALIZING;
        fireLifecycleEvent(Lifecycle.INIT_EVENT, null);
        try {
            ClassLoader loader = getLoader();
            Class<?> clazz = loader.loadClass(servletClass);
            servletInstance = (Servlet) clazz.getDeclaredConstructor().newInstance();
            servletInstance.init();
        } catch (Exception e) { e.printStackTrace(); }
        state = LifecycleState.INITIALIZED;
    }

    @Override
    public void start() { state = LifecycleState.STARTING; fireLifecycleEvent(Lifecycle.START_EVENT, null); state = LifecycleState.STARTED; }
    @Override
    public void stop() { state = LifecycleState.STOPPING; fireLifecycleEvent(Lifecycle.STOP_EVENT, null);
        if (servletInstance != null) servletInstance.destroy();
        state = LifecycleState.STOPPED; }
    @Override
    public void destroy() { state = LifecycleState.DESTROYING; fireLifecycleEvent(Lifecycle.DESTROY_EVENT, null); state = LifecycleState.DESTROYED; }

    public void invokeServlet(Request request, Response response) throws Exception {
        Context context = (Context) getParent();
        ApplicationFilterChain chain = new ApplicationFilterChain();
        for (String filterName : context.findFilters()) {
            Filter filter = context.findFilter(filterName);
            if (filter != null) chain.addFilter(filter);
        }
        chain.setServlet(servletInstance);
        chain.doFilter(request, response);
    }
}

package com.framework.tomcat.test;

import com.framework.tomcat.util.ServletMapping;

import java.io.*;
import java.net.ServerSocket;
import java.net.Socket;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

/**
 * 连接器，负责监听指定端口，接收 HTTP 请求并分发给对应 Servlet。
 */
public class Connector {

    private final int port;
    private final ExecutorService threadPool;
    private volatile boolean running;

    public Connector(int port) {
        this.port = port;
        this.threadPool = Executors.newCachedThreadPool();
    }

    public void start() {
        running = true;
        try (ServerSocket serverSocket = new ServerSocket(port)) {
            System.out.println("Connector: 启动监听端口 " + port);
            while (running) {
                Socket socket = serverSocket.accept();
                threadPool.submit(() -> process(socket));
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    private void process(Socket socket) {
        try (InputStream input = socket.getInputStream();
             OutputStream output = socket.getOutputStream()) {

            // 1. 解析请求
            Request request = new Request(input);
            request.parse();

            // 2. 创建响应对象
            Response response = new Response(output);

            // 3. 根据 URI 找到对应的 Servlet
            String uri = request.getUri();
            String servletClassName = ServletMapping.getServletClass(uri);
            if (servletClassName != null) {
                // 通过反射实例化 Servlet 并调用 service 方法
                Class<?> servletClass = Class.forName(servletClassName);
                Servlet servlet = (Servlet) servletClass.getDeclaredConstructor().newInstance();
                servlet.service(request, response);
            } else {
                // 未找到映射，返回 404
                response.setStatus(404, "Not Found");
                response.write("<h1>404 - Not Found</h1>");
            }

            // 4. 发送响应
            response.send();

        } catch (Exception e) {
            e.printStackTrace();
        } finally {
            try {
                socket.close();
            } catch (IOException e) {
                e.printStackTrace();
            }
        }
    }

    public void stop() {
        running = false;
        threadPool.shutdown();
    }
}

import java.io.InputStream;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;

/**
 * 封装 HTTP 请求，解析请求行和头部（本 Demo 只简单解析 URI）。
 */
public class Request {

    private final InputStream input;
    private String uri;
    private String method;

    public Request(InputStream input) {
        this.input = input;
    }

    public void parse() throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(input));
        String requestLine = reader.readLine();
        if (requestLine != null && !requestLine.isEmpty()) {
            String[] parts = requestLine.split(" ");
            if (parts.length >= 2) {
                method = parts[0];
                uri = parts[1];
            }
        }
        // 忽略剩余的 headers 和 body（简化处理）
        while (reader.readLine() != null) {
            // 空行表示 headers 结束，实际应用中应继续解析直到空行
            String line = reader.readLine();
            if (line == null || line.isEmpty()) break;
        }
    }

    public String getUri() {
        return uri;
    }

    public String getMethod() {
        return method;
    }
}

import java.io.OutputStream;

import java.io.IOException;
import java.nio.charset.StandardCharsets;

/**
 * 封装 HTTP 响应，支持写入内容并发送。
 */
public class Response {

    private final OutputStream output;
    private int statusCode = 200;
    private String statusMessage = "OK";
    private StringBuilder content = new StringBuilder();

    public Response(OutputStream output) {
        this.output = output;
    }

    public void setStatus(int code, String message) {
        this.statusCode = code;
        this.statusMessage = message;
    }

    public void write(String text) {
        content.append(text);
    }

    public void send() throws IOException {
        String contentStr = content.toString();
        byte[] contentBytes = contentStr.getBytes(StandardCharsets.UTF_8);

        PrintWriter writer = new PrintWriter(output, true);
        // 响应行
        writer.print("HTTP/1.1 " + statusCode + " " + statusMessage + "\r\n");
        // 响应头
        writer.print("Content-Type: text/html; charset=utf-8\r\n");
        writer.print("Content-Length: " + contentBytes.length + "\r\n");
        writer.print("\r\n");
        writer.flush();

        // 响应体
        output.write(contentBytes);
        output.flush();
    }
}

package com.framework.tomcat.util;

import java.io.File;
import java.net.URL;
import java.util.Enumeration;
import java.util.HashSet;
import java.util.Set;
import java.util.jar.JarEntry;
import java.util.jar.JarFile;

public class PackageScanner {

    /**
     * 扫描匹配模式（支持 * 和 **）的包下的所有类
     *
     * @param pattern 示例：com.codestats.*.controller 或 com.codestats.**
     * @return 匹配的类集合
     */
    public static Set<Class<?>> scan(String pattern) throws Exception {
        String basePackage = getBasePackage(pattern);
        Set<Class<?>> allClasses = getClassesRecursive(basePackage);
        Set<Class<?>> result = new HashSet<>();

        // 构造正则：将 * 替换为 [^.]+，** 替换为 .*
        String regex = buildRegex(pattern);

        for (Class<?> clazz : allClasses) {
            if (clazz.getName().matches(regex)) {
                result.add(clazz);
            }
        }
        return result;
    }

    // 从模式构建正则表达式
    private static String buildRegex(String pattern) {
        // 先转义点号，再处理通配符（注意顺序：先占位 **，再处理单 *）
        String regex = pattern
                .replace(".", "\\.")
                .replace("**", "##STARSTAR##")   // 占位
                .replace("*", "[^.]+")
                .replace("##STARSTAR##", ".*");
        // 匹配包括最后一层具体类名
        return "^" + regex + "(\\.[^.]+)?$";
    }

    // 提取基础包名（取第一个通配符前的部分）
    private static String getBasePackage(String pattern) {
        int starIdx = pattern.indexOf('*');
        if (starIdx == -1) {
            starIdx = pattern.indexOf('?');
        }
        if (starIdx == -1) {
            return pattern;
        }
        // 截取到最后一个完整的包段（即通配符前最近的 '.' 之后）
        String prefix = pattern.substring(0, starIdx);
        if (prefix.endsWith(".")) {
            prefix = prefix.substring(0, prefix.length() - 1);
        }
        return prefix;
    }

    // 递归扫描包及其子包（文件系统与JAR）
    private static Set<Class<?>> getClassesRecursive(String packageName) throws Exception {
        Set<Class<?>> classes = new HashSet<>();
        String path = packageName.replace('.', '/');
        Enumeration<URL> resources = Thread.currentThread().getContextClassLoader().getResources(path);
        while (resources.hasMoreElements()) {
            URL resource = resources.nextElement();
            if ("file".equals(resource.getProtocol())) {
                File dir = new File(resource.getFile());
                collectClassesFromDir(dir, packageName, classes);
            } else if ("jar".equals(resource.getProtocol())) {
                // 处理 jar 包中的类
                String jarPath = resource.getPath().substring(5, resource.getPath().indexOf("!"));
                try (JarFile jar = new JarFile(jarPath)) {
                    Enumeration<JarEntry> entries = jar.entries();
                    while (entries.hasMoreElements()) {
                        JarEntry entry = entries.nextElement();
                        String name = entry.getName();
                        if (name.endsWith(".class") && name.startsWith(path)) {
                            String className = name.substring(0, name.length() - 6).replace('/', '.');
                            addClassSafely(classes, className);
                        }
                    }
                }
            }
        }
        return classes;
    }

    private static void collectClassesFromDir(File dir, String packageName, Set<Class<?>> classes) {
        if (!dir.exists()) return;
        File[] files = dir.listFiles();
        if (files == null) return;
        for (File file : files) {
            if (file.isDirectory()) {
                collectClassesFromDir(file, packageName + "." + file.getName(), classes);
            } else if (file.getName().endsWith(".class")) {
                String className = packageName + "." + file.getName().substring(0, file.getName().length() - 6);
                addClassSafely(classes, className);
            }
        }
    }

    // 安全加载类，忽略任何加载/初始化异常
    private static void addClassSafely(Set<Class<?>> classes, String className) {
        try {
            classes.add(Class.forName(className));
        } catch (Throwable ignored) {
            // 忽略无法加载的类（如缺失依赖、初始化错误等）
        }
    }

    // 兼容旧方法（支持逗号分隔的多个模式或固定包名）
    public static Set<Class<?>> getClasses(String... packageNames) throws Exception {
        Set<Class<?>> allClasses = new HashSet<>();
        for (String pkg : packageNames) {
            if (pkg.contains("*") || pkg.contains("?")) {
                allClasses.addAll(scan(pkg));
            } else {
                allClasses.addAll(getClassesRecursive(pkg));
            }
        }
        return allClasses;
    }
}

package com.framework.util;

import com.framework.jdbc.metedata.datamodel.ColumnDef;
import com.framework.jdbc.metedata.datamodel.DataModel;
import com.framework.jdbc.metedata.datamodel.Row;

import java.util.*;
import java.util.stream.Collectors;

/**
 * DataModel 多场景转换工具，基于 JsonUtil 实现 JSON 序列化，
 * 同时支持 CSV、HTML、Markdown 等常用格式。
 */
public class DataModelConverter {

    /**
     * 将完整 DataModel（包含列定义和所有行）转换为 JSON 对象字符串。
     * 格式：{"columns":[...], "rows":[...]}
     */
    public static String toFullJson(DataModel dm) {
        Map<String, Object> root = new LinkedHashMap<>();
        root.put("columns", columnsToJson(dm));
        root.put("rows", rowsToJsonArray(dm, null)); // 所有列
        root.put("rowCount", dm.getRowCount());
        return JsonUtil.toJson(root);
    }

    /**
     * 将列定义转换为 JSON 数组，每个元素包含列名、类型等信息。
     */
    public static String columnsToJson(DataModel dm) {
        List<Map<String, Object>> list = new ArrayList<>();
        for (ColumnDef col : dm.getColumnDefs()) {
            Map<String, Object> map = new LinkedHashMap<>();
            map.put("name", col.getName());
            map.put("typeName", col.getTypeName());
            map.put("sqlType", col.getSqlType());
            map.put("nullable", col.isNullable());
            map.put("columnSize", col.getColumnSize());
            map.put("decimalDigits", col.getDecimalDigits());
            map.put("remarks", col.getRemarks());
            list.add(map);
        }
        return JsonUtil.toJson(list);
    }

    /**
     * 将所有行数据转换为 JSON 数组，每行是一个 JSON 对象，键为列名。
     */
    public static String rowsToJsonArray(DataModel dm) {
        return rowsToJsonArray(dm, null);
    }

    /**
     * 将指定列集的行数据转换为 JSON 数组。
     *
     * @param dm           数据模型
     * @param columnNames  需要包含的列名集合，为 null 或空则包含所有列
     */
    public static String rowsToJsonArray(DataModel dm, Set<String> columnNames) {
        List<Map<String, Object>> rows = new ArrayList<>();
        List<String> effectiveColNames = resolveColumnNames(dm, columnNames);

        for (int i = 0; i < dm.getRowCount(); i++) {
            Map<String, Object> rowMap = new LinkedHashMap<>();
            Row row = dm.getRow(i);
            for (String colName : effectiveColNames) {
                Object value = row.getValue(colName);
                rowMap.put(colName, value);
            }
            rows.add(rowMap);
        }
        return JsonUtil.toJson(rows);
    }

    /**
     * 将数据模型转换为 CSV 字符串（不含列定义，纯数据）。
     * 第一行为表头，后续为数据行。
     */
    public static String toCsv(DataModel dm) {
        return toCsv(dm, null);
    }

    /**
     * 将指定列集的数据模型转换为 CSV 字符串。
     */
    public static String toCsv(DataModel dm, Set<String> columnNames) {
        List<String> effectiveCols = resolveColumnNames(dm, columnNames);
        StringBuilder sb = new StringBuilder();

        // 表头
        sb.append(effectiveCols.stream().map(DataModelConverter::escapeCsv).collect(Collectors.joining(",")));
        sb.append("\n");

        // 数据行
        for (int i = 0; i < dm.getRowCount(); i++) {
            Row row = dm.getRow(i);
            String line = effectiveCols.stream()
                    .map(name -> {
                        Object val = row.getValue(name);
                        return val == null ? "" : escapeCsv(val.toString());
                    })
                    .collect(Collectors.joining(","));
            sb.append(line).append("\n");
        }
        return sb.toString();
    }

    /**
     * 转换为 HTML 表格字符串（基础样式）。
     */
    public static String toHtmlTable(DataModel dm) {
        return toHtmlTable(dm, null, "");
    }

    /**
     * 转换为 HTML 表格，支持自定义列和附加 CSS 类名。
     */
    public static String toHtmlTable(DataModel dm, Set<String> columnNames, String tableClass) {
        List<String> effectiveCols = resolveColumnNames(dm, columnNames);
        StringBuilder sb = new StringBuilder();
        sb.append("<table");
        if (tableClass != null && !tableClass.isEmpty()) {
            sb.append(" class='").append(tableClass).append("'");
        }
        sb.append(" border='1' cellspacing='0' cellpadding='5'>");

        // 表头
        sb.append("<thead><tr>");
        for (String col : effectiveCols) {
            sb.append("<th>").append(col).append("</th>");
        }
        sb.append("</tr></thead><tbody>");

        // 数据行
        for (int i = 0; i < dm.getRowCount(); i++) {
            sb.append("<tr>");
            Row row = dm.getRow(i);
            for (String col : effectiveCols) {
                Object val = row.getValue(col);
                sb.append("<td>").append(val != null ? val.toString() : "").append("</td>");
            }
            sb.append("</tr>");
        }
        sb.append("</tbody></table>");
        return sb.toString();
    }

    /**
     * 转换为 Markdown 表格。
     */
    public static String toMarkdownTable(DataModel dm) {
        return toMarkdownTable(dm, null);
    }

    /**
     * 转换为 Markdown 表格，支持自定义列。
     */
    public static String toMarkdownTable(DataModel dm, Set<String> columnNames) {
        List<String> effectiveCols = resolveColumnNames(dm, columnNames);
        StringBuilder sb = new StringBuilder();

        // 表头
        sb.append("| ");
        sb.append(String.join(" | ", effectiveCols));
        sb.append(" |\n");

        // 分隔线
        sb.append("| ");
        sb.append(effectiveCols.stream().map(s -> "---").collect(Collectors.joining(" | ")));
        sb.append(" |\n");

        // 数据行
        for (int i = 0; i < dm.getRowCount(); i++) {
            Row row = dm.getRow(i);
            sb.append("| ");
            String line = effectiveCols.stream()
                    .map(name -> {
                        Object val = row.getValue(name);
                        return val == null ? "" : val.toString();
                    })
                    .collect(Collectors.joining(" | "));
            sb.append(line).append(" |\n");
        }
        return sb.toString();
    }

    /* ---------- 内部辅助方法 ---------- */

    private static List<String> resolveColumnNames(DataModel dm, Set<String> specifiedColumns) {
        if (specifiedColumns != null && !specifiedColumns.isEmpty()) {
            // 按照传入顺序返回，但保持与 model 中列定义顺序一致
            List<String> ordered = new ArrayList<>();
            for (String col : dm.getColumnNames()) {
                if (specifiedColumns.contains(col)) {
                    ordered.add(col);
                }
            }
            return ordered;
        } else {
            return Arrays.asList(dm.getColumnNames());
        }
    }

    private static String escapeCsv(String value) {
        if (value.contains(",") || value.contains("\"") || value.contains("\n")) {
            return "\"" + value.replace("\"", "\"\"") + "\"";
        }
        return value;
    }
}

package com.framework.util;

import java.lang.reflect.Array;
import java.util.Collection;
import java.util.Map;

/**
 * 轻量级 JSON 序列化工具，模仿主流 JSON 库的用法。
 * 支持 String、Number、Boolean、null、Map、Collection、数组以及嵌套对象。
 */
public class JsonUtil {

    /**
     * 将任意 Java 对象序列化为 JSON 字符串。
     * @param obj 要序列化的对象
     * @return JSON 字符串
     */
    public static String toJson(Object obj) {
        StringBuilder sb = new StringBuilder();
        serialize(obj, sb);
        return sb.toString();
    }

    @SuppressWarnings("unchecked")
    private static void serialize(Object obj, StringBuilder sb) {
        if (obj == null) {
            sb.append("null");
        } else if (obj instanceof String) {
            sb.append('"').append(escape((String) obj)).append('"');
        } else if (obj instanceof Number || obj instanceof Boolean) {
            sb.append(obj.toString());
        } else if (obj instanceof Map) {
            serializeMap((Map<String, Object>) obj, sb);
        } else if (obj instanceof Collection) {
            serializeCollection((Collection<?>) obj, sb);
        } else if (obj.getClass().isArray()) {
            serializeArray(obj, sb);
        } else {
            // 其他复杂对象，暂时转为字符串表示（可扩展为反射获取字段）
            sb.append('"').append(escape(obj.toString())).append('"');
        }
    }

    private static void serializeMap(Map<String, Object> map, StringBuilder sb) {
        sb.append('{');
        boolean first = true;
        for (Map.Entry<String, Object> entry : map.entrySet()) {
            if (!first) {
                sb.append(',');
            }
            sb.append('"').append(escape(entry.getKey())).append("\":");
            serialize(entry.getValue(), sb);
            first = false;
        }
        sb.append('}');
    }

    private static void serializeCollection(Collection<?> coll, StringBuilder sb) {
        sb.append('[');
        boolean first = true;
        for (Object item : coll) {
            if (!first) {
                sb.append(',');
            }
            serialize(item, sb);
            first = false;
        }
        sb.append(']');
    }

    private static void serializeArray(Object array, StringBuilder sb) {
        sb.append('[');
        int len = Array.getLength(array);
        for (int i = 0; i < len; i++) {
            if (i > 0) {
                sb.append(',');
            }
            serialize(Array.get(array, i), sb);
        }
        sb.append(']');
    }

    /**
     * JSON 字符串转义（处理反斜杠、双引号、换行等）
     */
    private static String escape(String s) {
        StringBuilder sb = new StringBuilder(s.length());
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            switch (c) {
                case '\\': sb.append("\\\\"); break;
                case '"':  sb.append("\\\""); break;
                case '\n': sb.append("\\n"); break;
                case '\r': sb.append("\\r"); break;
                case '\t': sb.append("\\t"); break;
                default:   sb.append(c);
            }
        }
        return sb.toString();
    }
}
